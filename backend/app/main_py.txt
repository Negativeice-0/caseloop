from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
from typing import List
from datetime import datetime

from app.core.database import engine, Base, get_db
from app.models import models
from app.schemas import schemas
from app.api import admin


# Create database tables automatically on startup
Base.metadata.create_all(bind=engine)

app = FastAPI(
    title="CaseLoop API",
    description="Offline-first humanitarian response coordination backend for Red Cross branch context.",
    version="0.1.0",
    author="Billy Gates Osodo Ochieng"
)

app.include_router(admin.router)

# --- CORS BLOCK ---
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],  # Allows all local ports (Vite 5173, etc.)
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/")
def read_root():
    return {
        "app": "CaseLoop",
        "status": "online",
        "mode": "offline-first local database active",
        "version": "0.1.0"
    }

@app.get("/ping")
def ping():
    return {"ping": "pong"}

@app.post("/api/responses/", response_model=schemas.ResponseOut, status_code=status.HTTP_201_CREATED)
def create_response(response_in: schemas.ResponseCreate, coordinator_id: int = 1, db: Session = Depends(get_db)):
    existing = db.query(models.ResponseItem).filter(models.ResponseItem.reference_code == response_in.reference_code).first()
    if existing:
        raise HTTPException(status_code=400, detail="Response with this reference code already exists.")
    
    db_response = models.ResponseItem(
        reference_code=response_in.reference_code,
        title=response_in.title,
        description=response_in.description,
        category=response_in.category,
        priority=response_in.priority,
        location=response_in.location,
        coordinator_id=coordinator_id,
        status=models.ResponseStatus.ACTIVE
    )
    db.add(db_response)
    db.commit()
    db.refresh(db_response)

    event = models.ResponseEvent(
        response_id=db_response.id,
        actor_id=coordinator_id,
        event_type="CREATED",
        details=f"Response record created with title: {db_response.title}"
    )
    db.add(event)
    db.commit()

    return db_response

@app.get("/api/responses/", response_model=List[schemas.ResponseOut])
def list_responses(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
    return db.query(models.ResponseItem).offset(skip).limit(limit).all()

@app.get("/api/responses/{response_id}", response_model=schemas.ResponseOut)
def get_response(response_id: int, db: Session = Depends(get_db)):
    response = db.query(models.ResponseItem).filter(models.ResponseItem.id == response_id).first()
    if not response:
        raise HTTPException(status_code=404, detail="Response not found.")
    return response

@app.post("/api/assignments/", response_model=schemas.AssignmentOut, status_code=status.HTTP_201_CREATED)
def create_assignment(assignment_in: schemas.AssignmentCreate, db: Session = Depends(get_db)):
    resp = db.query(models.ResponseItem).filter(models.ResponseItem.id == assignment_in.response_id).first()
    if not resp:
        raise HTTPException(status_code=404, detail="Associated response not found.")
    
    db_assignment = models.Assignment(
        assignment_reference=assignment_in.assignment_reference,
        response_id=assignment_in.response_id,
        assignee_id=assignment_in.assignee_id,
        description=assignment_in.description,
        expected_completion=assignment_in.expected_completion,
        status=models.AssignmentStatus.ASSIGNED
    )
    db.add(db_assignment)
    db.commit()
    db.refresh(db_assignment)

    event = models.ResponseEvent(
        response_id=assignment_in.response_id,
        actor_id=assignment_in.assignee_id,
        event_type="ASSIGNMENT_ISSUED",
        details=f"Assignment {assignment_in.assignment_reference} issued."
    )
    db.add(event)
    db.commit()

    return db_assignment

@app.get("/api/responses/{response_id}/timeline", response_model=List[schemas.EventOut])
def get_response_timeline(response_id: int, db: Session = Depends(get_db)):
    return db.query(models.ResponseEvent).filter(models.ResponseEvent.response_id == response_id).order_by(models.ResponseEvent.timestamp.desc()).all()
