import React, { useEffect, useState } from 'react';
import { fetchResponses, createResponse, checkBackendHealth, requestAdminKeyword, verifyAdminKeyword } from './services/api';
import { ResponseItem } from './types';
import { ShieldAlert, PlusCircle, Activity, CheckCircle2, AlertCircle, Clock, Users, Search, RefreshCw, Settings } from 'lucide-react';
import { AdminControl } from './features/admin/AdminControl';

export default function App() {
  const [isAdminUnlocked, setIsAdminUnlocked] = useState(false);
  const [showAdminModal, setShowAdminModal] = useState(false);
  const [adminKeywordInput, setAdminKeywordInput] = useState('');

  const [activeTab, setActiveTab] = useState<'responses' | 'assignments' | 'timeline' | 'admin'>('responses');
  const [responses, setResponses] = useState<ResponseItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [backendStatus, setBackendStatus] = useState<'online' | 'offline'>('offline');
  const [error, setError] = useState<string | null>(null);
  const [searchTerm, setSearchTerm] = useState('');

  const [showForm, setShowForm] = useState(false);
  const [refCode, setRefCode] = useState(`CR-KL-2026-004`);
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [category, setCategory] = useState('Community assistance');
  const [priority, setPriority] = useState('normal');
  const [location, setLocation] = useState("Karen-Lang'ata Branch");
  const [submitting, setSubmitting] = useState(false);

  const loadData = async () => {
    try {
      setLoading(true);
      const health = await checkBackendHealth();
      if (health.status === 'online' || health.app === 'CaseLoop') {
        setBackendStatus('online');
      }
      const data = await fetchResponses();
      setResponses(data);
      setError(null);
    } catch (err: any) {
      setBackendStatus('offline');
      setError("Backend API is offline. Run 'uvicorn app.main:app --reload --port 8000' in your backend venv.");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadData();
  }, []);

  const handleRequestKeyword = async () => {
    try {
      const data = await requestAdminKeyword();
      alert(`🔑 Secure Keyword Generated!\n\nFor local demo testing, your key is: ${data.keyword}\n(Valid for ${data.expires_in_minutes} minutes)`);
    } catch (err: any) {
      alert(err.message);
    }
  };

  const handleVerifyKeyword = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      await verifyAdminKeyword(adminKeywordInput);
      setIsAdminUnlocked(true);
      setShowAdminModal(false);
      setAdminKeywordInput('');
      setActiveTab('admin');
      alert('Admin access granted successfully.');
    } catch (err: any) {
      alert(err.message);
    }
  };

  const handleCreate = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      setSubmitting(true);
      await createResponse({
        reference_code: refCode,
        title,
        description,
        category,
        priority,
        location,
      });
      setTitle('');
      setDescription('');
      setShowForm(false);
      loadData();
    } catch (err: any) {
      alert(err.message);
    } finally {
      setSubmitting(false);
    }
  };

  const filteredResponses = responses.filter(r => 
    r.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
    r.reference_code.toLowerCase().includes(searchTerm.toLowerCase()) ||
    r.category.toLowerCase().includes(searchTerm.toLowerCase())
  );

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 flex flex-col md:flex-row font-sans selection:bg-red-600 selection:text-white">
      
      {/* Sidebar Navigation */}
      <aside className="w-full md:w-72 bg-white border-r border-slate-200 p-6 flex flex-col justify-between shadow-xl relative z-10">
        <div>
          <div className="flex items-center gap-3 mb-8 group cursor-pointer">
            <div className="bg-red-600 p-3 rounded-2xl text-white shadow-lg shadow-red-600/30">
              <ShieldAlert className="w-6 h-6 animate-spin" style={{ animationDuration: '10s' }} />
            </div>
            <div>
              <h1 className="font-black text-xl tracking-tight text-slate-900 flex items-center gap-1.5">
                Case<span className="text-red-600">Loop</span>
              </h1>
              <p className="text-xs font-medium text-slate-500">Karen-Lang'ata Operations</p>
            </div>
          </div>

          <nav className="space-y-2">
            <button
              onClick={() => setActiveTab('responses')}
              className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-semibold transition-all ${
                activeTab === 'responses' ? 'bg-red-600 text-white shadow-md shadow-red-600/25' : 'text-slate-600 hover:bg-slate-100'
              }`}
            >
              <Activity className="w-4 h-4" /> Response Register
            </button>
            <button
              onClick={() => setActiveTab('assignments')}
              className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-semibold transition-all ${
                activeTab === 'assignments' ? 'bg-red-600 text-white shadow-md shadow-red-600/25' : 'text-slate-600 hover:bg-slate-100'
              }`}
            >
              <Users className="w-4 h-4" /> Assignments & Teams
            </button>
            <button
              onClick={() => setActiveTab('timeline')}
              className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-semibold transition-all ${
                activeTab === 'timeline' ? 'bg-red-600 text-white shadow-md shadow-red-600/25' : 'text-slate-600 hover:bg-slate-100'
              }`}
            >
              <Clock className="w-4 h-4" /> Operational Timeline
            </button>

            {isAdminUnlocked ? (
              <button
                onClick={() => setActiveTab('admin')}
                className={`w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-semibold transition-all ${
                  activeTab === 'admin' ? 'bg-slate-900 text-white shadow-md shadow-slate-900/25' : 'text-slate-600 hover:bg-slate-100'
                }`}
              >
                <Settings className="w-4 h-4" /> Admin & System Control
              </button>
            ) : (
              <button
                onClick={() => setShowAdminModal(true)}
                className="w-full flex items-center gap-3 px-4 py-3 rounded-xl text-sm font-semibold text-slate-500 hover:bg-slate-100 transition border border-dashed border-slate-300 mt-4"
              >
                <Settings className="w-4 h-4" /> 🔒 Unlock Admin
              </button>
            )}
          </nav>
        </div>

        <div className="bg-slate-50 border border-slate-200 p-4 rounded-2xl space-y-2.5 shadow-inner">
          <div className="flex items-center justify-between text-xs font-semibold">
            <span className="text-slate-500">System Mode</span>
            <span className="flex items-center gap-1.5 text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-full border border-emerald-200">
              <span className="w-2 h-2 rounded-full bg-emerald-500 animate-ping"></span> Offline-First
            </span>
          </div>
          <div className="flex items-center justify-between text-xs text-slate-600 pt-2 border-t border-slate-200 font-medium">
            <span>FastAPI Backend:</span>
            <span className={backendStatus === 'online' ? 'text-emerald-600 font-bold uppercase' : 'text-amber-600 font-bold uppercase'}>
              {backendStatus}
            </span>
          </div>
        </div>
      </aside>

      {/* Main Content Area */}
      <main className="flex-1 p-6 md:p-10 overflow-y-auto bg-slate-50">
        
        <header className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-8 pb-6 border-b border-slate-200">
          <div>
            <h2 className="text-2xl font-black tracking-tight text-slate-900 capitalize">
              {activeTab === 'responses' && 'Operational Response Register'}
              {activeTab === 'assignments' && 'Team Assignment Management'}
              {activeTab === 'timeline' && 'Chronological Operational Timeline'}
              {activeTab === 'admin' && 'Admin & System Resource Control'}
            </h2>
          </div>
          <div className="flex items-center gap-3">
            <button
              onClick={loadData}
              className="p-2.5 rounded-xl bg-white border border-slate-200 hover:bg-slate-100 text-slate-700 shadow-sm transition active:scale-95"
            >
              <RefreshCw className="w-4 h-4" />
            </button>
            {activeTab === 'responses' && (
              <button
                onClick={() => setShowForm(!showForm)}
                className="inline-flex items-center justify-center gap-2 bg-red-600 hover:bg-red-700 text-white font-bold px-5 py-2.5 rounded-xl shadow-lg shadow-red-600/30 text-sm"
              >
                <PlusCircle className="w-4 h-4" /> New Response
              </button>
            )}
          </div>
        </header>

        {error && (
          <div className="mb-6 p-4 rounded-2xl bg-amber-50 border border-amber-200 text-amber-800 flex items-center gap-3 text-sm shadow-sm">
            <AlertCircle className="w-5 h-5 flex-shrink-0 text-amber-600" />
            <span className="font-medium">{error}</span>
          </div>
        )}

        {/* TAB 1: RESPONSES */}
        {activeTab === 'responses' && (
          <>
            {showForm && (
              <div className="mb-8 bg-white border border-slate-200 p-6 rounded-2xl shadow-xl">
                <h3 className="text-base font-black mb-4 text-slate-900">Create New Operational Response Record</h3>
                <form onSubmit={handleCreate} className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-bold text-slate-700 mb-1 uppercase">Reference Code</label>
                    <input type="text" required value={refCode} onChange={(e) => setRefCode(e.target.value)} className="w-full bg-slate-50 border border-slate-300 rounded-xl px-3.5 py-2 text-sm" />
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-slate-700 mb-1 uppercase">Response Title</label>
                    <input type="text" required placeholder="e.g., Karen Medical Camp" value={title} onChange={(e) => setTitle(e.target.value)} className="w-full bg-slate-50 border border-slate-300 rounded-xl px-3.5 py-2 text-sm" />
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-slate-700 mb-1 uppercase">Category</label>
                    <select value={category} onChange={(e) => setCategory(e.target.value)} className="w-full bg-slate-50 border border-slate-300 rounded-xl px-3.5 py-2 text-sm">
                      <option value="Community assistance">Community assistance</option>
                      <option value="Disaster response">Disaster response</option>
                      <option value="Emergency support">Emergency support</option>
                    </select>
                  </div>
                  <div>
                    <label className="block text-xs font-bold text-slate-700 mb-1 uppercase">Priority</label>
                    <select value={priority} onChange={(e) => setPriority(e.target.value)} className="w-full bg-slate-50 border border-slate-300 rounded-xl px-3.5 py-2 text-sm">
                      <option value="normal">Normal</option>
                      <option value="high">High</option>
                      <option value="urgent">Urgent</option>
                    </select>
                  </div>
                  <div className="md:col-span-2">
                    <label className="block text-xs font-bold text-slate-700 mb-1 uppercase">Description</label>
                    <textarea rows={2} value={description} onChange={(e) => setDescription(e.target.value)} className="w-full bg-slate-50 border border-slate-300 rounded-xl px-3.5 py-2 text-sm"></textarea>
                  </div>
                  <div className="md:col-span-2 flex justify-end gap-3">
                    <button type="button" onClick={() => setShowForm(false)} className="px-4 py-2 rounded-xl bg-slate-200 text-sm font-bold">Cancel</button>
                    <button type="submit" disabled={submitting} className="px-5 py-2 rounded-xl bg-red-600 text-white text-sm font-bold shadow-md">Save Record</button>
                  </div>
                </form>
              </div>
            )}

            <div className="mb-6 flex items-center gap-3 bg-white border border-slate-200 px-4 py-3 rounded-2xl shadow-sm">
              <Search className="w-4 h-4 text-slate-400" />
              <input type="text" placeholder="Search responses..." value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} className="w-full bg-transparent text-sm focus:outline-none" />
            </div>

            <div className="bg-white border border-slate-200 rounded-2xl overflow-hidden shadow-xl">
              <div className="px-6 py-4 border-b border-slate-200 flex justify-between bg-slate-50">
                <h3 className="font-black text-sm text-slate-800 uppercase">Records ({filteredResponses.length})</h3>
                <span className="text-xs font-bold text-red-600 bg-red-50 border border-red-200 px-2 py-0.5 rounded font-mono">SQLite DB</span>
              </div>
              {loading ? (
                <div className="p-12 text-center text-sm text-slate-500 animate-pulse">Loading local records...</div>
              ) : (
                <div className="overflow-x-auto">
                  <table className="w-full text-left text-sm">
                    <thead className="bg-slate-100 text-xs uppercase font-bold text-slate-600">
                      <tr>
                        <th className="px-6 py-3">Reference</th>
                        <th className="px-6 py-3">Title</th>
                        <th className="px-6 py-3">Category</th>
                        <th className="px-6 py-3">Priority</th>
                        <th className="px-6 py-3">Status</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-200">
                      {filteredResponses.map((item) => (
                        <tr key={item.id} className="hover:bg-red-50/40">
                          <td className="px-6 py-4 font-mono text-xs font-black text-red-600">{item.reference_code}</td>
                          <td className="px-6 py-4 font-bold text-slate-900">{item.title}</td>
                          <td className="px-6 py-4 text-slate-600 text-xs">{item.category}</td>
                          <td className="px-6 py-4 uppercase text-xs font-bold">{item.priority}</td>
                          <td className="px-6 py-4 text-emerald-700 text-xs font-bold">{item.status}</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              )}
            </div>
          </>
        )}

        {activeTab === 'assignments' && <div className="bg-white p-10 rounded-2xl shadow-xl text-center">Team Assignments Module Active</div>}
        {activeTab === 'timeline' && <div className="bg-white p-10 rounded-2xl shadow-xl text-center">Operational Timeline Module Active</div>}
        {activeTab === 'admin' && isAdminUnlocked && <AdminControl />}
      </main>

      {/* Secure Admin Unlock Modal */}
      {showAdminModal && (
        <div className="fixed inset-0 bg-slate-950/60 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className="bg-white border border-slate-200 p-6 rounded-3xl shadow-2xl max-w-md w-full">
            <h3 className="text-lg font-black text-slate-900 mb-1 flex items-center gap-2">
              <Settings className="w-5 h-5 text-red-600" /> Restricted Admin Access
            </h3>
            <p className="text-xs text-slate-600 mb-6 font-medium">Enter a valid dynamic keyword to unlock system controls.</p>

            <div className="mb-5 p-3.5 bg-slate-50 border border-slate-200 rounded-2xl flex items-center justify-between">
              <div>
                <p className="text-[11px] font-bold text-slate-700">Need an access key?</p>
                <p className="text-[10px] text-slate-500">Request a fresh single-use keyword.</p>
              </div>
              <button
                type="button"
                onClick={handleRequestKeyword}
                className="px-3 py-1.5 bg-slate-900 hover:bg-slate-800 text-white text-xs font-bold rounded-xl shadow"
              >
                Request Key
              </button>
            </div>

            <form onSubmit={handleVerifyKeyword} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-slate-700 mb-1 uppercase tracking-wider">Access Keyword</label>
                <input
                  type="text"
                  required
                  placeholder="e.g., CR-KL-XXXXXX"
                  value={adminKeywordInput}
                  onChange={(e) => setAdminKeywordInput(e.target.value)}
                  className="w-full bg-slate-50 border border-slate-300 rounded-xl px-4 py-2.5 text-sm font-mono text-slate-900 uppercase"
                />
              </div>

              <div className="flex justify-end gap-3 pt-2">
                <button
                  type="button"
                  onClick={() => setShowAdminModal(false)}
                  className="px-4 py-2.5 rounded-xl bg-slate-200 text-sm font-bold text-slate-700"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl bg-red-600 hover:bg-red-700 text-sm font-bold text-white shadow-md shadow-red-600/30"
                >
                  Unlock System
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
